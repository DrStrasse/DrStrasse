# --- Чат-команды вида "!команда аргументы" от владельца ----------------------
event chat(Ply:entity, Msg:string, Team:number) {
    if (Ply != Owner | Msg[1] != "!") { return }

    let Args = Msg:sub(2):explode(" ")
    switch (Args[1, string]) {
        case "ping",
            hideChat(1)
            print("pong")
            break

        case "say",
            hideChat(1)
            print(Args:concat(" ", 2))   # текст с пробелами, начиная со 2-го слова
            break

        default,
            print("Команды: !ping, !say <текст>")
    }
}
